    @PostMapping("/register")
    public String registerUser(@RequestBody User user){
        User existingUSer=userRepository.findByEmail(user.getEmail());
        if(existingUSer!=null){
            return "User already exists";
        }
        user.setPassword(passwordEncoder.encode(user.getPassword()));
        if(user.getRole()==null || user.getRole().isEmpty()){
            user.setRole("ROLE_USER");
        }
        userRepository.save(user);
        return "Registration Successful";
    }
    //======================LOGIN=========================
    @PostMapping("/login")
    public LoginResponse login(@RequestBody User user){
        User existingUser=userRepository.findByEmail(user.getEmail());
        if(existingUser==null){
            throw new RuntimeException("User not found");}
        if(!existingUser.isEnabled()){
            throw new RuntimeException("User is disabled by admin");
        }
        if(passwordEncoder.matches(user.getPassword(),existingUser.getPassword())){
            String token = JwtUtil.generateToken(existingUser.getEmail(),existingUser.getRole());
            return new LoginResponse("Bearer "+token,existingUser.getEmail(),existingUser.getRole());
        }else{
            throw new RuntimeException("wrong password");
        }
    }

    //======================PROFILE PIC VIEW==============================
    @GetMapping("/profile/image/{filename}")
    public ResponseEntity<Resource> getProfileImage(@PathVariable String filename)throws IOException {
        Path path=Paths.get("uploads").resolve(filename).normalize();
        UrlResource resource=new UrlResource(path.toUri());
        if(!resource.exists() || !resource.isReadable()){
            throw new RuntimeException("File not found"+filename);
        }
        return  ResponseEntity.ok().body(resource);
    }
    //====================CHANGE/UPDATE PASSWORD=========================
    @PutMapping("/profile/password")
    public String changePassword(@RequestBody ChangePasswordRequest req) {
        String email = (String) SecurityContextHolder
                .getContext()
                .getAuthentication()
                .getPrincipal();
        User user = userRepository.findByEmail(email);
        if (user == null) {
            return "User not found";
        }
        if (!passwordEncoder.matches(req.getOldPassword(), user.getPassword())) {
            return "Old password is incorrect";
        }
        user.setPassword(passwordEncoder.encode(req.getNewPassword()));
        userRepository.save(user);
        return "Password changed";
    }
    //=====================UPLODE PROFILE PICTURE====================
    @PostMapping("/profile/upload/picture")
    public String uploadProfilepicture(@RequestParam("file") MultipartFile file){
        try{
            String email= (String) SecurityContextHolder
                    .getContext()
                    .getAuthentication()
                    .getPrincipal();
            User user = userRepository.findByEmail(email);
            if(user==null){
                return "User not found";
            }
            String fileName = System.currentTimeMillis()+"_"+file.getOriginalFilename();
            String uploadDir="uploads/";
            File uploadPath=new File(uploadDir);

            if(!uploadPath.exists()){
                uploadPath.mkdir();
            }

            Path filepath= Paths.get(uploadDir+fileName);
            Files.write(filepath,file.getBytes());

            user.setProfilePicture(fileName);
            userRepository.save(user);
            return "Profile picture uploaded"+fileName;
        }
        catch (Exception e){
            return "Upload Failed";
        }
    }

    @GetMapping("/books/search")
    public Page<BookResponse> searchBook(@RequestParam String title,
                                         Pageable pageable) {

        Page<Book> books =
                bookRepository.findByTitleContainingIgnoreCaseAndStatus(
                        title, "AVAILABLE", pageable);

        return books.map(book -> {
            BookResponse dto = new BookResponse();
            dto.setTitle(book.getTitle());
            dto.setAuthor(book.getAuthor());
            dto.setPrice(book.getPrice());
            dto.setDescription(book.getDescription());

            User user = book.getUser();
            if (user != null) {
                dto.setSellerName(user.getName());
                dto.setSellerPhone(user.getPhone());
            }
            return dto;
        });
    }

    @PutMapping("/seller/book/update/{id}")
    public String updateBook(@PathVariable int id, @RequestBody Book Updatebook){
        Book book=bookRepository.findById(id).orElse(null);
        if(book==null){
            return "book not found";
        }
        String email=(String) getContext().getAuthentication().getPrincipal();
        User currentUser=userRepository.findByEmail(email);
        if(book.getUser().getId()!= currentUser.getId()){
            return "Not allowed to update";
        }
        book.setTitle(Updatebook.getTitle());
        book.setAuthor(Updatebook.getAuthor());
        book.setPrice(Updatebook.getPrice());
        book.setDescription(Updatebook.getDescription());
        bookRepository.save(book);
        return "Updated";
    }

    @PostMapping("/order/buy/{bookId}")
    public String buyBook(@PathVariable int bookId){
        String email=SecurityContextHolder
//        User loggeduser=(User) SecurityContextHolder
                .getContext()
                .getAuthentication()
                .getName();
//        String email=loggeduser.getEmail();

        User buyer=userRepository.findByEmail(email);
        Book book=bookRepository.findById(bookId).orElse(null);

        if(book==null){
            throw new RuntimeException("Book not found");
        }
        if(!"AVAILABLE".equals(book.getStatus())){
            throw new RuntimeException("Book is not available");
        }
        if(book.getUser().getId()==buyer.getId()){
            throw new RuntimeException("you can not buy your own book");
        }
        if(book.getQuantity()<=0){
            throw new RuntimeException("out of stock");
        }
        Order order=new Order();
        order.setBuyer(buyer);
        order.setBook(book);
        order.setQuantity(1);
        order.setPrice(book.getPrice());
        order.setStatus("PLACED");
        orderRepository.save(order);
        book.setQuantity(book.getQuantity()-1);
//      return new OrderResponse(order.getId(),"Order placed successfully");
/*        OrderResponse res=new OrderResponse();
        res.setOrderId(order.getId());
        res.setMessage("Order placed successfully");*/
        if(book.getQuantity()==0){
            book.setStatus("SOLD");
        }
        bookRepository.save(book);
        return "Order placed Successfully";
    }

    @PostMapping("/review/{bookId}")
    public String reviewBook(@PathVariable int bookId,@RequestBody ReviewRequest req){
        String email=(String) SecurityContextHolder
                .getContext()
                .getAuthentication()
                .getPrincipal();
        User user=userRepository.findByEmail(email);
        Book book=bookRepository.findById(bookId).orElse(null);

        if(book==null){
            return "Book not found";
        }
        if(req.getRating()<1 || req.getRating()>5){
            return "Rating must be between 1 and 5";
        }
        Review review=new Review();
        review.setRating(req.getRating());
        review.setComments(req.getComment());
        review.setUser(user);
        review.setBook(book);

        reviewRepository.save(review);
        return "Review Added Successfully";
    }

    @DeleteMapping("/user/cart/remove")
    public String removeitem(@RequestParam int bookId,@RequestParam int quantity){
        String email=(String) SecurityContextHolder
                .getContext()
                .getAuthentication()
                .getPrincipal();

        Cart cart=cartRepository.findByUserEmailAndBookId(email,bookId);
        if(cart==null){
            return "Item not found";
        }
        if(quantity<=0){
            return "Quantity must be greater than 0";
        }
        if(cart.getQuantity()<quantity){
            return "Quantity must be greater than or equal to quantity";
        }
        int remaning=cart.getQuantity()-quantity;
        if(remaning>0){
            cart.setQuantity(remaning);
            cartRepository.save(cart);
        }else {
            cartRepository.delete(cart);
        }
        return "item remove and updeted";
    }

    //=============================CHECK OUT===================================
    @PostMapping("/user/checkout")
    @Transactional
    public String checkout(){
        String email=SecurityContextHolder
                .getContext()
                .getAuthentication()
                .getName();
        List<Cart> cartList=cartRepository.findByUserEmail(email);

        if(cartList.isEmpty()){
            return "Cart is Empty";
        }
        for(Cart cart:cartList){
            Book book=bookRepository.findById(cart.getBookId()).orElse(null);
            if(book==null){
                continue;
            }
            if(book.getQuantity()<cart.getQuantity()){
                return "Not enough stock";
            }
            book.setQuantity(book.getQuantity()-cart.getQuantity());
            bookRepository.save(book);
            User user=userRepository.findByEmail(email);
            Order order=new Order();
            order.setBuyer(user);
            order.setBook(book);
            order.setQuantity(cart.getQuantity());
            order.setPrice(book.getPrice()*cart.getQuantity());

            order.setStatus("PLACED");
            orderRepository.save(order);
        }
        return "order placed successfully";
    }
